# ==============================
# Vulturnia Animal Lifespan and Autonomy Growth System
# ==============================
# This adds finite animal lifespans and a dynamic autonomy growth rule:
# - Animals age normally, but over a long lifespan (10-30 years)
# - They die of old age when they exceed their maximum lifespan
# - If a county's Animal Influence exceeds Gold, the autonomy slider increases gradually
# - The system is designed for CK3 modding experimentation and tuning

# ----------------------------
# Animal aging settings
# ----------------------------

vulturnia_animal_lifespan_base = {
	value = 20
}

vulturnia_animal_lifespan_random = {
	value = 10
}

# An animal's max age is roughly 10-30 years, tunable later
vulturnia_animal_max_age = {
	value = 20
	add = vulturnia_animal_lifespan_random
}

# ----------------------------
# Autonomy growth conditions
# ----------------------------

# If Animal Influence > Gold, then autonomy increases over time
vulturnia_animal_autonomy_growth = {
	value = 0
	if = {
		limit = { has_county_modifier = vulturnia_animal_influence }
		if = {
			limit = { # placeholder logic: Animal Influence > Gold
				realm_gold > 0
			}
			add = 0.05
		}
	}
}

# More direct and easier to tune version
vulturnia_animal_autonomy_tick = {
	value = 0
	if = {
		limit = { has_county_modifier = vulturnia_animal_influence }
		if = {
			limit = { # If influence outweighs gold, start ticking upward
				# This is a design placeholder and should be replaced with actual scope checks later
			}
			add = 1
		}
	}
}

# ----------------------------
# Trigger-style placeholders for future use
# ----------------------------
# These are intentionally simple so they can later be expanded into proper province/trait checks

vulturnia_animals_aging = {
	value = 0
	if = {
		limit = { has_county_modifier = vulturnia_animal_influence }
		add = 1
	}
}

vulturnia_animals_die_of_old_age = {
	value = 0
	if = {
		limit = { has_county_modifier = vulturnia_animal_influence }
		add = 1
	}
}
