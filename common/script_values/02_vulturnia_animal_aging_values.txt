# ==============================
# Vulturnia Animal Aging & Autonomy Growth
# ==============================
# This is a cleaner, safer scaffold for the Animal Influence lifecycle.
# The goal is to keep the design clear while avoiding impossible scope tricks.
#
# Rules:
# - Animal lifespan is intentionally long: roughly 10-30 years
# - Old age death is a natural event, not a hard-coded combat mechanic
# - If Animal Influence exceeds Gold, autonomy grows gradually over time
# - Autonomy is represented as a county-level policy change, not a direct resource bar

# ----------------------------
# Lifespan values
# ----------------------------

vulturnia_animal_lifespan_min = {
	value = 10
}

vulturnia_animal_lifespan_max = {
	value = 30
}

vulturnia_animal_lifespan_average = {
	value = 20
}

# ----------------------------
# Autonomy growth values
# ----------------------------
# These are intentionally conservative and easy to tune.
# The design intent is that high Animal Influence gradually pushes the county toward autonomy.

vulturnia_animal_autonomy_gain = {
	value = 0
	if = {
		limit = { has_county_modifier = vulturnia_animal_influence }
		add = 0.50
	}
}

vulturnia_animal_autonomy_growth = {
	value = 0
	if = {
		limit = {
			has_county_modifier = vulturnia_animal_influence
			NOT = { has_county_modifier = vulturnia_animal_full_control }
		}
		add = 1.00
	}
}

# ----------------------------
# Old-age placeholder trigger
# ----------------------------
# This is a simple placeholder meant to be hooked into a more advanced animal lifecycle system later.

vulturnia_animals_old_age = {
	value = 0
	if = {
		limit = { has_county_modifier = vulturnia_animal_influence }
		add = 1
	}
}

vulturnia_animals_die_of_old_age = {
	value = 0
	if = {
		limit = { has_county_modifier = vulturnia_animal_influence }
		add = 1
	}
}
