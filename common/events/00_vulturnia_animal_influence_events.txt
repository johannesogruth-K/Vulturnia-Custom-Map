# Event: Lion Hunt Success
event = {
	id = vulturnia_animal.0002
	title = vulturnia_animal.0002.t
	description = vulturnia_animal.0002.desc
	picture = "gfx/events/character_interactions/hunt.dds"

	trigger = {
		religion = aquis
		any_held_title = {
			has_holding_type = temple
			holding = { has_building = building_vulturnia_lion_hunting_grounds }
		}
	}

	immediate = {
		# Success in hunting provides prestige and martial experience
	}

	option = {
		name = vulturnia_animal.0002.a
		add_prestige = 100
		add_martial_lifestyle_xp = 50
	}

	option = {
		name = vulturnia_animal.0002.b
		add_gold = 75
	}
}
